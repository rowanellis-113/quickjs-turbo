cmake_minimum_required(VERSION 3.20)
project(QuickJSObjectNullHost C)
find_package(qjs CONFIG REQUIRED)
option(QJS_OBJECT_NULL_NATIVE_COUNTERS "Assert private WIN64 static native route coverage" OFF)
add_executable(interpreter-object-null main.c)
target_link_libraries(interpreter-object-null PRIVATE qjs)
enable_testing()
if(QJS_OBJECT_NULL_NATIVE_COUNTERS)
    get_target_property(qjs_type qjs TYPE)
    if(NOT WIN32 OR NOT qjs_type STREQUAL "STATIC_LIBRARY")
        message(
            FATAL_ERROR
                "Native counters require a static WIN64 engine; use default API checks otherwise")
    endif()
    target_compile_definitions(interpreter-object-null PRIVATE OBJECT_NULL_NATIVE_COUNTERS)
    if(MINGW)
        target_compile_definitions(interpreter-object-null PRIVATE OBJECT_NULL_MINGW)
    endif()
    set(on_count 1)
else()
    set(on_count -1)
endif()
foreach(mode on off all-loose-off interpreter-off)
    if(mode STREQUAL "on")
        set(expected ${on_count})
        set(extra "")
    else()
        set(expected 0)
        if(mode STREQUAL "off")
            set(extra "QJS_DISABLE_FRAME_LOOSE_EQ_OBJECT_NULL=1")
        elseif(mode STREQUAL "all-loose-off")
            set(extra "QJS_DISABLE_FRAME_LOOSE_EQ=1")
        else()
            set(extra "QJS_DISABLE_INTERP_FRAME=1")
        endif()
    endif()
    add_test(NAME object-null-${mode} COMMAND interpreter-object-null ${expected})
    set_tests_properties(
        object-null-${mode}
        PROPERTIES ENVIRONMENT "QJS_EXPERIMENT_ALLOCATOR=256;QJS_OBSERVE_FRAME_LOOSE_EQ=1;${extra}")
endforeach()
