cmake_minimum_required(VERSION 3.20)
project(CharacterLeafPublicHost C)
find_package(qjs CONFIG REQUIRED)
add_executable(character-host main.c)
target_compile_features(character-host PRIVATE c_std_11)
target_link_libraries(character-host PRIVATE qjs)
if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
    target_compile_options(character-host PRIVATE -g -gdwarf-4)
endif()
enable_testing()
foreach(mode native disabled noic register-c c)
    set(extra "")
    if(mode STREQUAL "disabled")
        set(extra "QJS_DISABLE_FRAME_CHARACTER_SET_LEAF=0")
    elseif(mode STREQUAL "noic")
        set(extra "QJS_DISABLE_PROPERTY_IC=1")
    elseif(mode STREQUAL "register-c")
        set(extra "QJS_DISABLE_REGISTER_INTERP=1")
    elseif(mode STREQUAL "c")
        set(extra
            "QJS_DISABLE_INTERP_FRAME=1;QJS_DISABLE_INTERNAL_INTERP=1;QJS_DISABLE_REGISTER_INTERP=1"
        )
    endif()
    add_test(NAME character-${mode} COMMAND character-host)
    set_tests_properties(character-${mode} PROPERTIES ENVIRONMENT
                                                      "QJS_EXPERIMENT_ALLOCATOR=256;${extra}")
endforeach()
