cmake_minimum_required(VERSION 3.20)
project(MethodICInstalledPublicHosts C)
find_package(qjs CONFIG REQUIRED)
if(NOT FRAME_SOURCE_ROOT)
    message(FATAL_ERROR "FRAME_SOURCE_ROOT must identify the frozen source fixtures")
endif()
add_executable(method-host main.c)
add_executable(compound-host "${FRAME_SOURCE_ROOT}/tests/common/embedding/compound-leaf/main.c")
add_executable(property-host "${FRAME_SOURCE_ROOT}/tests/common/native_property_ic_public.c")
foreach(target method-host compound-host property-host)
    target_compile_features(${target} PRIVATE c_std_11)
    target_link_libraries(${target} PRIVATE qjs)
endforeach()
enable_testing()
foreach(mode native compound-off method-off ic-off register-c c)
    set(extra "")
    if(mode STREQUAL "compound-off")
        set(extra "QJS_DISABLE_FRAME_COMPOUND_LEAF=0")
    elseif(mode STREQUAL "method-off")
        set(extra "QJS_DISABLE_METHOD_IC_LEARNING=0")
    elseif(mode STREQUAL "ic-off")
        set(extra "QJS_DISABLE_PROPERTY_IC=0")
    elseif(mode STREQUAL "register-c")
        set(extra "QJS_DISABLE_REGISTER_INTERP=1")
    elseif(mode STREQUAL "c")
        set(extra
            "QJS_DISABLE_INTERP_FRAME=1;QJS_DISABLE_INTERNAL_INTERP=1;QJS_DISABLE_REGISTER_INTERP=1"
        )
    endif()
    add_test(NAME method-${mode}
             COMMAND method-host "${FRAME_SOURCE_ROOT}/tests/common/test_property_ic.js"
                     "${FRAME_SOURCE_ROOT}/tests/common/test_vm_prototype_chain.js")
    add_test(NAME compound-${mode} COMMAND compound-host)
    add_test(NAME property-${mode} COMMAND property-host)
    foreach(kind method compound property)
        set_tests_properties(${kind}-${mode} PROPERTIES ENVIRONMENT
                                                        "QJS_EXPERIMENT_ALLOCATOR=256;${extra}")
    endforeach()
endforeach()
