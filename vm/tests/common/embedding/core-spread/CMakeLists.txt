cmake_minimum_required(VERSION 3.20)
project(CoreSpreadPublicHost C)
find_package(qjs CONFIG REQUIRED)
add_executable(core-spread-host main.c)
target_compile_features(core-spread-host PRIVATE c_std_11)
target_link_libraries(core-spread-host PRIVATE qjs)
enable_testing()
foreach(mode native disabled noic register-c c)
    set(extra "")
    if(mode STREQUAL "disabled")
        set(extra "QJS_DISABLE_CORE_SPREAD_CALL=0")
    elseif(mode STREQUAL "noic")
        set(extra "QJS_DISABLE_PROPERTY_IC=1")
    elseif(mode STREQUAL "register-c")
        set(extra "QJS_DISABLE_REGISTER_INTERP=1")
    elseif(mode STREQUAL "c")
        set(extra
            "QJS_DISABLE_INTERP_FRAME=1;QJS_DISABLE_INTERNAL_INTERP=1;QJS_DISABLE_REGISTER_INTERP=1"
        )
    endif()
    add_test(NAME core-spread-${mode} COMMAND core-spread-host)
    set_tests_properties(core-spread-${mode} PROPERTIES ENVIRONMENT
                                                        "QJS_EXPERIMENT_ALLOCATOR=256;${extra}")
endforeach()
