cmake_minimum_required(VERSION 3.20)
project(RegExpTestPublicSafety C)
find_package(qjs CONFIG REQUIRED)
add_executable(regexp-test-public main.c)
target_compile_features(regexp-test-public PRIVATE c_std_11)
if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
    target_compile_options(regexp-test-public PRIVATE -g -gdwarf-4)
endif()
target_link_libraries(regexp-test-public PRIVATE qjs)
enable_testing()
foreach(mode default noic register-c c)
    set(extra "")
    if(mode STREQUAL "noic")
        set(extra "QJS_DISABLE_PROPERTY_IC=1")
    elseif(mode STREQUAL "register-c")
        set(extra "QJS_DISABLE_REGISTER_INTERP=1")
    elseif(mode STREQUAL "c")
        set(extra
            "QJS_DISABLE_INTERP_FRAME=1;QJS_DISABLE_INTERNAL_INTERP=1;QJS_DISABLE_REGISTER_INTERP=1"
        )
    endif()
    add_test(NAME regexp-test-${mode} COMMAND regexp-test-public)
    set_tests_properties(regexp-test-${mode} PROPERTIES ENVIRONMENT
                                                        "QJS_EXPERIMENT_ALLOCATOR=256;${extra}")
endforeach()
