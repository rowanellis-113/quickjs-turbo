cmake_minimum_required(VERSION 3.20)
project(BindingCellPublicHost C)
find_package(qjs CONFIG REQUIRED)
add_executable(binding-cell-host main.c)
target_compile_features(binding-cell-host PRIVATE c_std_11)
target_link_libraries(binding-cell-host PRIVATE qjs)
enable_testing()
foreach(mode native disabled noic register-c c)
    set(extra "")
    if(mode STREQUAL "disabled")
        set(extra "QJS_DISABLE_FRAME_BINDING_CELL=0")
    elseif(mode STREQUAL "noic")
        set(extra "QJS_DISABLE_PROPERTY_IC=1")
    elseif(mode STREQUAL "register-c")
        set(extra "QJS_DISABLE_REGISTER_INTERP=1")
    elseif(mode STREQUAL "c")
        set(extra
            "QJS_DISABLE_INTERP_FRAME=1;QJS_DISABLE_INTERNAL_INTERP=1;QJS_DISABLE_REGISTER_INTERP=1"
        )
    endif()
    add_test(NAME binding-cell-${mode} COMMAND binding-cell-host)
    set_tests_properties(binding-cell-${mode} PROPERTIES ENVIRONMENT
                                                         "QJS_EXPERIMENT_ALLOCATOR=256;${extra}")
endforeach()
